async function shareSelectedPhotosToWhatsApp(event) { const p = projects.find(x => x.id === currentProjectId); if (selectedPhotos.size === 0) return alert("Veuillez sélectionner au moins une photo."); const btn = event.target; const originalText = btn.textContent; btn.textContent = "Préparation..."; btn.disabled = true; try { const now = new Date(); const optionsDate = { year: 'numeric', month: '2-digit', day: '2-digit' }; const dateStr = now.toLocaleDateString(currentLang === 'en' ? 'en-GB' : currentLang, optionsDate).replace(/\//g, '-'); const timeStr = now.toTimeString().slice(0,5).replace(":", "h"); const files = []; const selectedIndices = Array.from(selectedPhotos).sort((a,b) => a-b); for (const idx of selectedIndices) { const dataUrl = p.photoData[idx]; if (!dataUrl) continue; const bin = atob(dataUrl.split(",")[1]); const arr = new Uint8Array(bin.length); for (let i = 0; i < bin.length; i++) arr[i] = bin.charCodeAt(i); const filename = `${p.name}_${dateStr}_${timeStr}_P${idx+1}.jpg`; files.push(new File([arr], filename, { type: "image/jpeg" })); } // Préparer le texte et le copier dans le presse-papiers const shareText = `${t('project')}: ${p.name}\n${t('date')}: ${dateStr} ${t('at')} ${timeStr}`; await navigator.clipboard.writeText(shareText); // IMPORTANT : On envoie UNIQUEMENT les fichiers, sans le champ 'text' const shareData = { files: files // 'text' et 'title' sont retirés pour éviter le bug de WhatsApp }; if (navigator.canShare && navigator.canShare(shareData)) { await navigator.share(shareData); // Message après le partage pour rappeler de coller le texte setTimeout(() => { alert("Les photos ont été envoyées.\n\nLe texte (Projet + Date) a été copié. Dans WhatsApp, faites un appui long dans la barre de message et cliquez sur 'Coller'."); }, 500); } else { alert("Votre téléphone ne supporte pas le partage direct de plusieurs fichiers."); } } catch (e) { if (e.name !== 'AbortError') { console.error(e); alert("Erreur lors du partage : " + e.message); } } finally { btn.textContent = originalText; btn.disabled = false; } }
